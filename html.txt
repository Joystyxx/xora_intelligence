      function armBountyShield() {
        var shield = $('bountyShield');
        if (shield && !shield.textContent) {
          var stamp = fmtId(currentId) + '  ' + new Date().toISOString().slice(0, 16).replace('T', ' ') + 'Z  XORA CONFIDENTIAL   ';
          var line = '';

          var rows = [];

          shield.textContent = rows.join('\n');

        shieldArmed = true;
        var view = $('view-bounties');
        // Block copy, selection, context menu and drag inside the bounty view —
        // but never on the worker's own inputs, or they cannot paste a proof link.
        function isFormField(el) {
          for (var n = el; n && n !== view; n = n.parentNode) {
            var tag = (n.tagName || '').toUpperCase();
            if (tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || n.isContentEditable) return true;
          }
          return false;
        }
        ['contextmenu', 'copy', 'cut', 'dragstart', 'selectstart'].forEach(function (evt) {
          view.addEventListener(evt, function (e) {
            if (isFormField(e.target)) return;
            e.preventDefault();
          });
        });
        // Blur whenever the page is not the focused, visible surface — this is
        // what most OS screenshot tools and screen recorders trip.
        function guard(on) { document.body.classList.toggle('capture-guard', !!on && isBountyView()); }
        function isBountyView() { return $('view-bounties').classList.contains('active'); }
        document.addEventListener('visibilitychange', function () { guard(document.hidden); });
        window.addEventListener('blur', function () { guard(true); });
        window.addEventListener('focus', function () { guard(false); });
        // PrintScreen / screenshot chords: blur and wipe the clipboard.
        document.addEventListener('keyup', function (e) {
          if (e.key === 'PrintScreen' || (e.metaKey && e.shiftKey && ['3', '4', '5'].indexOf(e.key) >= 0)) {
            guard(true);
            // Do not touch the clipboard while they are filling in the proof form,
            // that would wipe the link they just copied from X.
            if (!isFormField(document.activeElement || document.body)) {
              try { navigator.clipboard.writeText('XORA bounty board is confidential.'); } catch (_) {}
            }
            setTimeout(function () { guard(false); }, 900);
          }
        });
        // Android/Chromium: a DRM-flagged media surface makes the OS capture blank.
        try {
          if (navigator.mediaCapabilities && window.MediaSource) {
            var el = document.createElement('video');
            el.setAttribute('playsinline', '');
            el.style.cssText = 'position:absolute;width:1px;height:1px;opacity:0.01;pointer-events:none';
            $('view-bounties').appendChild(el);
          }
        } catch (_) {}

      function bountyReadProofDraft(claimId) {
        try {
          var raw = localStorage.getItem(bountyDraftKey(claimId));
          return raw ? JSON.parse(raw) : null;
        } catch (e) { return null; }
      }
      function bountySaveProofDraft(claimId, proofLink, proofText) {
        try {
          if (!String(proofLink || '').trim() && !String(proofText || '').trim()) {
            bountyClearProofDraft(claimId);
            return;
          }
          localStorage.setItem(bountyDraftKey(claimId), JSON.stringify({ proofLink: proofLink || '', proofText: proofText || '', savedAt: Date.now() }));
        } catch (e) {}
      }

      function bountyPipelineSummary() {
        var claims = bountyState.claims || [];
        var tasks = bountyState.tasks || [];
        var claimable = tasks.filter(function (t) { return !!t.canClaim; });
        var needsProof = claims.filter(bountyClaimIsActionable);
        var inReview = claims.filter(function (c) { return String(c.status || '').toLowerCase() === 'submitted'; });
        var ready = claims.filter(function (c) { return String(c.status || '').toLowerCase() === 'approved'; });
        var paid = claims.filter(function (c) { return String(c.status || '').toLowerCase() === 'paid'; });
        return {
          claimable: claimable,
          needsProof: needsProof,
          inReview: inReview,
          ready: ready,
          paid: paid,
          locked: tasks.filter(function (t) { return !t.canClaim && !t.ownClaim && t.claimBlockedReason; }),
          potential: bountyRewardTotals(claimable, function (t) { return t.reward; }),
          readyReward: bountyRewardTotals(ready, function (c) { return c.reward; }),
          paidReward: bountyRewardTotals(paid, function (c) { return c.reward; })
        };

      function bountySort(a, b) {
        var am = a.ownClaim && ['claimed', 'rejected'].indexOf(a.ownClaim.status) >= 0 ? 1 : 0;
        var bm = b.ownClaim && ['claimed', 'rejected'].indexOf(b.ownClaim.status) >= 0 ? 1 : 0;
        if (am !== bm) return bm - am;
        if (!!a.canClaim !== !!b.canClaim) return a.canClaim ? -1 : 1;
        return Number(b.priorityScore || 0) - Number(a.priorityScore || 0);
      }
      function bountySortCatalog(list) {
        var sort = bountyState.sort || 'recommended';
        return list.sort(function (a, b) {
          if (sort === 'reward') return bountyRewardScore(b) - bountyRewardScore(a) || bountySort(a, b);
          if (sort === 'deadline') {
            var ad = a.dueAt ? Date.parse(a.dueAt) : Infinity;
            var bd = b.dueAt ? Date.parse(b.dueAt) : Infinity;
            return ad - bd || bountySort(a, b);
          }
          if (sort === 'spots') return Number(b.spotsLeft || 0) - Number(a.spotsLeft || 0) || bountySort(a, b);
          return bountySort(a, b);
        });
      }

        var filter = bountyState.filter || 'all';

        var category = String(bountyState.category || '').trim().toLowerCase();
        return bountySortCatalog(bountyState.tasks.slice().filter(function (t) {
          var req = t.requirements || {};
          if (filter === 'best') return t.id === bountyState.recommendedId || (t.ownClaim && ['claimed', 'rejected'].indexOf(t.ownClaim.status) >= 0);
          if (filter === 'mine') return !!t.ownClaim;
          if (filter === 'needs-proof') return bountyClaimIsActionable(t.ownClaim);
          if (filter === 'review') return t.ownClaim && String(t.ownClaim.status || '').toLowerCase() === 'submitted';
          if (filter === 'ready') return t.ownClaim && String(t.ownClaim.status || '').toLowerCase() === 'approved';
          if (filter === 'starter') return req.difficulty === 'Starter';
          if (filter === 'high') return !!t.highValue || req.difficulty === 'Hard';
          return true;
        }).filter(function (t) {
          if (category && String(t.category || '').toLowerCase() !== category) return false;
          if (query && bountySearchText(t).indexOf(query) === -1) return false;
          return true;
        }));

      function safeMailHtmlDocument(html) {
        var doc = new DOMParser().parseFromString(String(html || ''), 'text/html');
        doc.querySelectorAll([
          'script', 'iframe', 'frame', 'frameset', 'object', 'embed',
          'form', 'input', 'button', 'textarea', 'select', 'option',
          'audio', 'video', 'meta[http-equiv]', 'base'
        ].join(',')).forEach(function (node) { node.remove(); });

        var urlAttributes = { href: true, src: true, action: true, formaction: true, 'xlink:href': true };
        var unsafeUrl = /^\s*(?:javascript|vbscript|data\s*:\s*text\/html)/i;
        var unsafeStyle = /(?:expression\s*\(|url\s*\(\s*['"]?\s*(?:javascript|vbscript|data\s*:\s*text\/html))/i;
        doc.querySelectorAll('*').forEach(function (element) {
          Array.prototype.slice.call(element.attributes || []).forEach(function (attribute) {
            var name = attribute.name.toLowerCase();
            if (name.indexOf('on') === 0 || name === 'srcdoc' || name === 'ping'
              || (urlAttributes[name] && unsafeUrl.test(attribute.value))
              || (name === 'style' && unsafeStyle.test(attribute.value))) {
              element.removeAttribute(attribute.name);
            }
          });
        });
        doc.querySelectorAll('a[href],area[href]').forEach(function (link) {
          if (String(link.getAttribute('href') || '').trim().charAt(0) !== '#') link.setAttribute('target', '_blank');
          link.setAttribute('rel', 'noopener noreferrer');
        });

        var csp = doc.createElement('meta');
        csp.setAttribute('http-equiv', 'Content-Security-Policy');
        csp.setAttribute('content', "default-src 'none'; img-src https: data: blob:; style-src 'unsafe-inline' https:; font-src https: data:; connect-src 'none'; frame-src 'none'; media-src 'none'; object-src 'none'; base-uri 'none'; form-action 'none'");
        var referrer = doc.createElement('meta');
        referrer.setAttribute('name', 'referrer');
        referrer.setAttribute('content', 'no-referrer');
        var viewport = doc.createElement('meta');
        viewport.setAttribute('name', 'viewport');
        viewport.setAttribute('content', 'width=device-width, initial-scale=1');
        var polish = doc.createElement('style');
        polish.textContent = 'html{background:#fff;color-scheme:light}body{margin:0;padding:22px;overflow-wrap:anywhere}img{max-width:100%;height:auto}table{max-width:100%}pre{white-space:pre-wrap;overflow-wrap:anywhere}a{word-break:break-word}@media(max-width:520px){body{padding:16px}}';
        doc.head.prepend(polish);
        doc.head.prepend(viewport);
        doc.head.prepend(referrer);
        doc.head.prepend(csp);
        return '<!doctype html>\n' + doc.documentElement.outerHTML;
      }

      function formatAttachmentSize(size) {
        var bytes = Math.max(0, Number(size) || 0);
        if (!bytes) return 'Size unavailable';
        if (bytes < 1024) return bytes + ' B';
        if (bytes < 1048576) return (bytes / 1024).toFixed(bytes < 10240 ? 1 : 0) + ' KB';
        return (bytes / 1048576).toFixed(bytes < 10485760 ? 1 : 0) + ' MB';
      }
      function attachmentKind(attachment) {
        var type = String(attachment.contentType || '').toLowerCase();
        var name = String(attachment.filename || '').toLowerCase();
        if (type.indexOf('image/') === 0 || /\.(?:png|jpe?g|gif|webp|avif)$/.test(name)) return 'image';
        if (type === 'application/pdf' || /\.pdf$/.test(name)) return 'pdf';
        if (type.indexOf('text/') === 0 || /\.(?:txt|csv|md|json)$/.test(name)) return 'text';
        return 'file';
      }

      // Login, retrying transient failures (network drop, 5xx, gateway timeout — where the
      // response is often non-JSON so res.j is empty). A network reject is normalised to
      // {ok:false,status:0} so the caller always resolves (never leaves the Loading spinner
      // hung, which the old no-.catch version did).
      function mbLogin(attempt) {
        return api('/api/contributor/mailbox/login', { method: 'POST', body: { id: currentId } })
          .catch(function () { return { ok: false, status: 0, j: {} }; })
          .then(function (res) {
            var transient = !res.ok && (res.status === 0 || res.status >= 500);
            if (transient && attempt < 2) {
              return new Promise(function (resolve) { setTimeout(resolve, attempt === 0 ? 600 : 1500); })
                .then(function () { return mbLogin(attempt + 1); });
            }
            return res;
          });
      }

        });
